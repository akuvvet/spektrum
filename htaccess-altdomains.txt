# Weiterleitung der alten Domains auf spektrum-nachhilfe.de
# Diese Datei in das Webverzeichnis von spektrum-ev.de bzw.
# spektrum-sg.de legen — nicht auf die neue Domain.

RewriteEngine On

# --- HTTPS zuerst --------------------------------------------
RewriteCond %{HTTPS} !=on
RewriteCond %{HTTP:X-Forwarded-Proto} !https
RewriteRule ^(.*)$ https://%{HTTP_HOST}/$1 [R=301,L]

# --- Let's Encrypt weiter durchlassen ------------------------
# Sonst laeuft das Zertifikat der alten Domain aus und jede
# Weiterleitung endet in einer Sicherheitswarnung.
RewriteRule ^\.well-known/ - [L]
